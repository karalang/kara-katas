# history.db: a browser's history
read history.db :: SELECT u.title, datetime(v.visit_time / 1000000 - 11644473600, 'unixepoch') AS visited, v.transition & 255 AS core FROM visits v JOIN urls u ON u.id = v.url ORDER BY v.visit_time DESC LIMIT 8
read history.db parse_dates=t:us :: SELECT id, visit_time - 11644473600000000 AS t FROM visits ORDER BY id LIMIT 5
read history.db :: SELECT substr(url, instr(url, '//') + 2, instr(substr(url, instr(url, '//') + 2), '/') - 1) AS host, sum(visit_count) AS visits FROM urls GROUP BY host ORDER BY visits DESC, host
read history.db parse_dates=day :: SELECT date(visit_time / 1000000 - 11644473600, 'unixepoch') AS day, count(*) AS visits, round(sum(visit_duration) / 1e6 / 60, 1) AS minutes FROM visits GROUP BY day
read history.db :: SELECT CASE transition & 255 WHEN 0 THEN 'link' WHEN 1 THEN 'typed' WHEN 2 THEN 'bookmark' WHEN 5 THEN 'generated' ELSE 'other' END AS how, (transition & 805306368) != 0 AS chain, count(*) AS n FROM visits GROUP BY 1, 2 ORDER BY 1, 2
read history.db :: SELECT * FROM sqlite_sequence
read history.db :: SELECT id, coalesce(title, url) AS label, visit_count, typed_count, hidden FROM urls ORDER BY id
read history.db :: SELECT id, (visit_time - lag(visit_time) OVER (ORDER BY visit_time)) / 1000000 AS gap_s, lead(url, 1, -1) OVER (ORDER BY visit_time) AS next_url FROM visits ORDER BY visit_time LIMIT 6
read history.db :: SELECT url, visit_count, round(100.0 * visit_count / (SELECT sum(visit_count) FROM urls), 1) AS pct FROM urls WHERE url LIKE 'https://www.sqlite.org/%' OR url LIKE '%wikipedia%' ORDER BY visit_count DESC
