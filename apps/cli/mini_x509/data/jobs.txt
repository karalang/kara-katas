# mini_x509 jobs: each line is an openssl command run from data/
x509 -in root.pem -noout -text
x509 -in issuing.pem -noout -text
x509 -in shop.pem -noout -text
x509 -in client.pem -noout -text
x509 -in api-ec.pem -noout -text
x509 -in localhost.pem -noout -text
x509 -in shop.der -inform DER -noout -subject -issuer -serial -dates
x509 -in chain.pem -noout -subject -enddate
x509 -in shop.pem -noout -fingerprint -sha256
x509 -in root.pem -noout -fingerprint
x509 -in shop.pem -noout -ext subjectAltName,keyUsage
x509 -in shop.pem -noout -checkhost www.shop.kara.test
x509 -in shop.pem -noout -checkhost a.b.shop.kara.test
x509 -in shop.pem -noout -checkhost KARA.test
x509 -in shop.pem -noout -checkip 192.0.2.10
x509 -in shop.pem -noout -checkip 2001:db8::11
x509 -in client.pem -noout -checkemail bot@kara.test
x509 -in tampered.pem -noout -subject
x509 -in garbage.pem -noout -text
x509 -in notes.txt -noout -text
x509 -in missing.pem -noout -text
asn1parse -in localhost.pem -i
verify -attime 1792022400 -CAfile root.pem -untrusted issuing.pem shop.pem client.pem api-ec.pem
verify -attime 1792022400 -CAfile root.pem -untrusted issuing.pem old.pem
verify -attime 1792022400 -CAfile root.pem -untrusted issuing.pem tampered.pem
verify -attime 1792022400 -CAfile root.pem shop.pem
verify -attime 1792022400 -CAfile root.pem -untrusted issuing.pem -untrusted shop.pem rogue.pem
verify -attime 1792022400 -CAfile root.pem -untrusted issuing.pem -purpose sslserver client.pem
verify -attime 1792022400 -CAfile root.pem -untrusted issuing.pem -verify_hostname evil.test shop.pem
verify -attime 1792022400 -CAfile root.pem -untrusted issuing.pem -verify_hostname x.shop.kara.test shop.pem
verify -attime 1792022400 localhost.pem
