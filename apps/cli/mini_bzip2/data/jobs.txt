# mini_bzip2 jobs: each line runs from data/
bzip2 -tv access.log.bz2 big.txt.bz2 empty.bz2 concat.bz2
bzip2 -tvv big.txt.bz2
bzip2 -t trunc.bz2
bzip2 -t flipped.bz2
bzip2 -tv trailing.bz2
bzip2 -t notes.bz2
bzip2 -t missing.bz2
bzip2 -dc concat.bz2
bzip2 -dk big.txt.bz2
sha256sum big.txt
bzip2 -kv -9 poem.txt
sha256sum poem.txt.bz2
bzip2 -kv -1 runs.bin
sha256sum runs.bin.bz2
bzip2 -k empty.txt
sha256sum empty.txt.bz2
bzip2 -k access.log
bzip2 -kfv -6 access.log
sha256sum access.log.bz2
